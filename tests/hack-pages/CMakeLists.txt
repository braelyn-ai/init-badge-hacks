cmake_minimum_required(VERSION 3.16)
project(hack_page_preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Builds ONE hack page with the shared UI code and renders it off-device.
#   -DHACK_PAGE=eyes  ->  ui/page_eyes.cpp and its make_eyes() factory.
set(HACK_PAGE "" CACHE STRING "Page file suffix, e.g. eyes for ui/page_eyes.cpp")
if(NOT HACK_PAGE)
    message(FATAL_ERROR "Pass -DHACK_PAGE=<name>")
endif()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(FACTORY_COMPONENTS "${REPO_ROOT}/.build/factory-components" CACHE PATH "Pinned factory component sources")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
add_link_options(-fsanitize=address,undefined)
add_subdirectory("${FACTORY_COMPONENTS}/lvgl" lvgl)
set(UI "${REPO_ROOT}/firmware/factory_badge/main/ui")
file(GLOB FONTS "${UI}/font_*.c")
add_executable(hack_page_preview main.cpp "${UI}/page_${HACK_PAGE}.cpp" "${UI}/hack_kit.cpp"
    "${UI}/widgets.cpp" "${UI}/design_assets.cpp" "${UI}/brand_assets.cpp" ${FONTS})
target_compile_definitions(hack_page_preview PRIVATE HACK_FACTORY=make_${HACK_PAGE})
target_include_directories(hack_page_preview PRIVATE "${REPO_ROOT}/firmware/factory_badge/main")
target_link_libraries(hack_page_preview PRIVATE lvgl)
